<script>
const byId = id => document.getElementById(id);
const n = value => Number.isFinite(value) ? Math.round(value * 10) / 10 : '未知';
const buttons = [...document.querySelectorAll('nav button')];
const label = element => element?.closest('button')?.textContent || element?.tagName || '屏外';
function draw() {
  const vv = window.visualViewport;
  const inset = getComputedStyle(byId('safe'));
  const rect = buttons[0].getBoundingClientRect();
  const hits = buttons.map(button => {
    const box = button.getBoundingClientRect();
    return label(document.elementFromPoint(box.x + box.width / 2, box.y + box.height / 2));
  });
  const mode = window.top !== window ? '内嵌（仅供预览）' : navigator.standalone || matchMedia('(display-mode: standalone)').matches ? '主屏幕' : '浏览器';
  byId('metrics').textContent = [
    '入口  ' + mode + '   单位 CSS px',
    'clientHeight  ' + document.documentElement.clientHeight,
    'innerHeight  ' + innerHeight,
    'visualViewport  ' + n(vv?.height) + '   顶偏移 ' + n(vv?.offsetTop),
    '100dvh  ' + n(byId('dynamic').getBoundingClientRect().height),
    '绝对定位 100%  ' + n(byId('absolute').getBoundingClientRect().height),
    '固定定位上下沿  ' + n(byId('fixed').getBoundingClientRect().top) + ' – ' + n(byId('fixed').getBoundingClientRect().bottom),
    '安全区 上 / 下  ' + inset.paddingTop + ' / ' + inset.paddingBottom,
    '按钮上下沿  ' + n(rect.top) + ' – ' + n(rect.bottom),
    '中心命中  ' + hits.join(' / ')
  ].join('\n');
}
let touches = 0;
document.addEventListener('pointerdown', event => {
  touches++;
  byId('last').textContent = '触点 ' + touches + ' · ' + event.pointerType + ' (' + n(event.clientX) + ', ' + n(event.clientY) + ') · 实收 ' + label(event.target) + ' · 坐标命中 ' + label(document.elementFromPoint(event.clientX,event.clientY));
}, true);
buttons.forEach(button => button.addEventListener('click', () => { byId('last').textContent += ' · 已点击' + button.textContent; }));
for (const event of ['resize','pageshow','focus','scroll']) window.addEventListener(event,draw);
window.visualViewport?.addEventListener('resize',draw);
window.visualViewport?.addEventListener('scroll',draw);
document.addEventListener('visibilitychange',draw);
draw();
</script>
